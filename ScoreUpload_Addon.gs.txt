// WORSHIP-SCHEDULE Score Upload Add-on
// Apps Script 프로젝트의 새 .gs 파일에 그대로 붙여넣으세요.
// v3.4: 대용량 악보를 Google Drive resumable upload로 분할 전송합니다.

const SCORE_STORE = Object.freeze({
  APP: 'worship-score',
  FOLDER_PROP: 'WORSHIP_SCORE_FOLDER_ID',
  FILE_PROP: 'WORSHIP_SCORE_FILE_ID',
  NAME_PROP: 'WORSHIP_SCORE_FILE_NAME',
  MIME_PROP: 'WORSHIP_SCORE_FILE_MIME',
  UPDATED_PROP: 'WORSHIP_SCORE_UPDATED_AT',
  UPLOAD_PREFIX: 'WORSHIP_SCORE_UPLOAD_',
  FOLDER_NAME: 'WORSHIP-SCHEDULE_CURRENT_SCORE',
  CHUNK_SIZE: 768 * 1024,
  MAX_BYTES: 25 * 1024 * 1024
});

function doPost(e) {
  const p = (e && e.parameter) || {};
  const requestId = String(p.requestId || '');
  try {
    if (String(p.app || '') !== SCORE_STORE.APP) {
      return scoreReply_({ ok: false, error: '지원하지 않는 요청입니다.' }, requestId);
    }

    const action = String(p.action || '');
    if (action === 'meta') {
      return scoreReply_({ ok: true, file: getCurrentScoreMeta_() }, requestId);
    }
    if (action === 'uploadStart') {
      return scoreReply_(startScoreUpload_(p), requestId);
    }
    if (action === 'uploadChunk') {
      return scoreReply_(uploadScoreChunk_(p), requestId);
    }
    if (action === 'delete') {
      deleteCurrentScore_();
      return scoreReply_({ ok: true, file: null }, requestId);
    }

    return scoreReply_({ ok: false, error: '지원하지 않는 악보 작업입니다.' }, requestId);
  } catch (err) {
    console.error(err && err.stack ? err.stack : err);
    return scoreReply_({ ok: false, error: String(err && err.message ? err.message : err) }, requestId);
  }
}

function startScoreUpload_(p) {
  cleanupStaleScoreUploads_();

  const fileName = sanitizeScoreFileName_(String(p.fileName || '').trim());
  const mimeType = normalizeScoreMime_(String(p.mimeType || ''), fileName);
  const totalSize = Number(p.totalSize || 0);

  if (!fileName) throw new Error('파일명이 비어 있습니다.');
  if (!isAllowedScoreFile_(fileName, mimeType)) {
    throw new Error('PDF, PPT, PPTX, DOC, DOCX, JPG, JPEG, PNG 파일만 올릴 수 있습니다.');
  }
  if (!Number.isFinite(totalSize) || totalSize <= 0) throw new Error('파일 크기를 확인할 수 없습니다.');
  if (totalSize > SCORE_STORE.MAX_BYTES) throw new Error('파일은 25MB 이하만 업로드할 수 있습니다.');

  const folder = getScoreFolder_();
  const initUrl = 'https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&fields=id,name,mimeType,size';
  const resp = UrlFetchApp.fetch(initUrl, {
    method: 'post',
    contentType: 'application/json; charset=UTF-8',
    headers: {
      Authorization: 'Bearer ' + ScriptApp.getOAuthToken(),
      'X-Upload-Content-Type': mimeType,
      'X-Upload-Content-Length': String(totalSize)
    },
    payload: JSON.stringify({ name: fileName, parents: [folder.getId()] }),
    muteHttpExceptions: true
  });

  const code = resp.getResponseCode();
  if (code < 200 || code >= 300) {
    throw new Error('Drive 업로드 준비 실패 (' + code + '): ' + truncate_(resp.getContentText(), 300));
  }

  const sessionUrl = getHeaderIgnoreCase_(resp.getAllHeaders(), 'Location');
  if (!sessionUrl) throw new Error('Drive 업로드 세션을 만들지 못했습니다.');

  const uploadId = Utilities.getUuid();
  const meta = {
    uploadId: uploadId,
    sessionUrl: String(sessionUrl),
    fileName: fileName,
    mimeType: mimeType,
    totalSize: totalSize,
    nextStart: 0,
    createdAt: Date.now()
  };
  PropertiesService.getScriptProperties().setProperty(SCORE_STORE.UPLOAD_PREFIX + uploadId, JSON.stringify(meta));

  return { ok: true, uploadId: uploadId, chunkSize: SCORE_STORE.CHUNK_SIZE, totalSize: totalSize };
}

function uploadScoreChunk_(p) {
  const uploadId = String(p.uploadId || '');
  if (!uploadId) throw new Error('업로드 세션 정보가 없습니다.');

  const props = PropertiesService.getScriptProperties();
  const key = SCORE_STORE.UPLOAD_PREFIX + uploadId;
  const raw = props.getProperty(key);
  if (!raw) throw new Error('업로드 세션이 만료되었습니다. 다시 업로드해주세요.');

  const meta = JSON.parse(raw);
  if (meta.complete && meta.fileId) {
    const completedFile = DriveApp.getFileById(String(meta.fileId));
    return { ok: true, complete: true, receivedBytes: Number(meta.totalSize || 0), file: buildScoreMeta_(completedFile, String(meta.updatedAt || '')) };
  }
  const start = Number(p.start || 0);
  const end = Number(p.end || -1);
  const totalSize = Number(p.totalSize || meta.totalSize || 0);
  const dataBase64 = String(p.dataBase64 || '');

  if (!dataBase64) throw new Error('업로드 조각 데이터가 비어 있습니다.');
  if (totalSize !== Number(meta.totalSize)) throw new Error('파일 크기 정보가 일치하지 않습니다.');
  if (start < 0 || end < start || end >= totalSize) throw new Error('업로드 범위가 올바르지 않습니다.');

  // 이전 요청이 실제로 성공했지만 브라우저에서 응답만 놓친 경우 같은 조각 재전송을 성공으로 처리합니다.
  if (end < Number(meta.nextStart || 0)) {
    return { ok: true, complete: false, receivedBytes: Number(meta.nextStart || 0) };
  }
  if (start !== Number(meta.nextStart || 0)) {
    throw new Error('업로드 순서가 어긋났습니다. 다시 시도해주세요.');
  }

  const bytes = Utilities.base64Decode(dataBase64);
  const expectedLength = end - start + 1;
  if (bytes.length !== expectedLength) throw new Error('업로드 조각 크기가 일치하지 않습니다.');

  const resp = UrlFetchApp.fetch(String(meta.sessionUrl), {
    method: 'put',
    contentType: String(meta.mimeType || 'application/octet-stream'),
    headers: {
      'Content-Range': 'bytes ' + start + '-' + end + '/' + totalSize
    },
    payload: bytes,
    muteHttpExceptions: true
  });

  const code = resp.getResponseCode();
  if (code === 308) {
    meta.nextStart = end + 1;
    props.setProperty(key, JSON.stringify(meta));
    return { ok: true, complete: false, receivedBytes: meta.nextStart };
  }

  if (code >= 200 && code < 300) {
    let driveMeta = {};
    try { driveMeta = JSON.parse(resp.getContentText() || '{}'); } catch (e) {}
    const fileId = String(driveMeta.id || '');
    if (!fileId) throw new Error('Drive 업로드는 완료됐지만 파일 ID를 받지 못했습니다.');

    const file = DriveApp.getFileById(fileId);
    try {
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (sharingErr) {
      try { file.setTrashed(true); } catch (e) {}
      throw new Error('미리보기 공유 설정에 실패했습니다: ' + sharingErr.message);
    }

    const folder = getScoreFolder_();
    const files = folder.getFiles();
    while (files.hasNext()) {
      const f = files.next();
      if (f.getId() !== fileId) {
        try { f.setTrashed(true); } catch (e) {}
      }
    }

    const updatedAt = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss');
    props.setProperties({
      [SCORE_STORE.FILE_PROP]: fileId,
      [SCORE_STORE.NAME_PROP]: meta.fileName,
      [SCORE_STORE.MIME_PROP]: meta.mimeType,
      [SCORE_STORE.UPDATED_PROP]: updatedAt
    }, true);
    meta.complete = true;
    meta.fileId = fileId;
    meta.updatedAt = updatedAt;
    meta.nextStart = totalSize;
    props.setProperty(key, JSON.stringify(meta));

    return { ok: true, complete: true, receivedBytes: totalSize, file: buildScoreMeta_(file, updatedAt) };
  }

  throw new Error('Drive 업로드 실패 (' + code + '): ' + truncate_(resp.getContentText(), 300));
}

function cleanupStaleScoreUploads_() {
  const props = PropertiesService.getScriptProperties();
  const all = props.getProperties();
  const cutoff = Date.now() - 24 * 60 * 60 * 1000;
  Object.keys(all).forEach(function(key) {
    if (key.indexOf(SCORE_STORE.UPLOAD_PREFIX) !== 0) return;
    try {
      const meta = JSON.parse(all[key] || '{}');
      if (!meta.createdAt || Number(meta.createdAt) < cutoff) props.deleteProperty(key);
    } catch (e) {
      props.deleteProperty(key);
    }
  });
}

function deleteCurrentScore_() {
  const props = PropertiesService.getScriptProperties();
  const folder = getScoreFolder_();
  const files = folder.getFiles();
  while (files.hasNext()) {
    try { files.next().setTrashed(true); } catch (e) {}
  }
  props.deleteProperty(SCORE_STORE.FILE_PROP);
  props.deleteProperty(SCORE_STORE.NAME_PROP);
  props.deleteProperty(SCORE_STORE.MIME_PROP);
  props.deleteProperty(SCORE_STORE.UPDATED_PROP);
}

function getCurrentScoreMeta_() {
  const props = PropertiesService.getScriptProperties();
  const fileId = props.getProperty(SCORE_STORE.FILE_PROP) || '';
  if (!fileId) return null;

  try {
    const file = DriveApp.getFileById(fileId);
    if (file.isTrashed()) return null;
    return buildScoreMeta_(file, props.getProperty(SCORE_STORE.UPDATED_PROP) || '');
  } catch (e) {
    props.deleteProperty(SCORE_STORE.FILE_PROP);
    props.deleteProperty(SCORE_STORE.NAME_PROP);
    props.deleteProperty(SCORE_STORE.MIME_PROP);
    props.deleteProperty(SCORE_STORE.UPDATED_PROP);
    return null;
  }
}

function buildScoreMeta_(file, updatedAt) {
  const id = file.getId();
  return {
    id: id,
    name: file.getName(),
    mimeType: file.getMimeType(),
    size: file.getSize(),
    updatedAt: updatedAt || '',
    previewUrl: 'https://drive.google.com/file/d/' + encodeURIComponent(id) + '/preview',
    openUrl: 'https://drive.google.com/file/d/' + encodeURIComponent(id) + '/view'
  };
}

function getScoreFolder_() {
  const props = PropertiesService.getScriptProperties();
  const savedId = props.getProperty(SCORE_STORE.FOLDER_PROP) || '';
  if (savedId) {
    try {
      const folder = DriveApp.getFolderById(savedId);
      if (!folder.isTrashed()) return folder;
    } catch (e) {}
  }

  const folder = DriveApp.createFolder(SCORE_STORE.FOLDER_NAME);
  props.setProperty(SCORE_STORE.FOLDER_PROP, folder.getId());
  return folder;
}

function scoreReply_(data, requestId) {
  const payload = Object.assign({ source: 'worship-score-api', requestId: String(requestId || '') }, data || {});
  const json = JSON.stringify(payload).replace(/</g, '\\u003c');
  return HtmlService
    .createHtmlOutput('<!doctype html><meta charset="utf-8"><script>window.parent.postMessage(' + json + ', "*");</script>')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function sanitizeScoreFileName_(name) {
  return String(name || '').replace(/[\\/:*?"<>|]/g, '_').slice(0, 180);
}

function normalizeScoreMime_(mimeType, fileName) {
  const ext = String(fileName || '').toLowerCase().split('.').pop();
  const map = {
    pdf: 'application/pdf',
    ppt: 'application/vnd.ms-powerpoint',
    pptx: 'application/vnd.openxmlformats-officedocument.presentationml.presentation',
    doc: 'application/msword',
    docx: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    jpg: 'image/jpeg',
    jpeg: 'image/jpeg',
    png: 'image/png'
  };
  return map[ext] || mimeType || 'application/octet-stream';
}

function isAllowedScoreFile_(fileName, mimeType) {
  const ext = String(fileName || '').toLowerCase().split('.').pop();
  return ['pdf','ppt','pptx','doc','docx','jpg','jpeg','png'].includes(ext) && !!mimeType;
}

function getHeaderIgnoreCase_(headers, name) {
  const target = String(name || '').toLowerCase();
  const keys = Object.keys(headers || {});
  for (let i = 0; i < keys.length; i++) {
    if (String(keys[i]).toLowerCase() === target) return headers[keys[i]];
  }
  return '';
}

function truncate_(text, maxLen) {
  const s = String(text || '');
  return s.length > maxLen ? s.slice(0, maxLen) + '…' : s;
}
