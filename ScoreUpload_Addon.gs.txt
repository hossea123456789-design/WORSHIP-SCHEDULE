// WORSHIP-SCHEDULE Score Upload Add-on
// Apps Script 프로젝트의 새 .gs 파일에 그대로 붙여넣으세요.
// 기존 일정/편성 API와 독립적으로 doPost만 사용합니다.

const SCORE_STORE = Object.freeze({
  APP: 'worship-score',
  FOLDER_PROP: 'WORSHIP_SCORE_FOLDER_ID',
  FILE_PROP: 'WORSHIP_SCORE_FILE_ID',
  NAME_PROP: 'WORSHIP_SCORE_FILE_NAME',
  MIME_PROP: 'WORSHIP_SCORE_FILE_MIME',
  UPDATED_PROP: 'WORSHIP_SCORE_UPDATED_AT',
  FOLDER_NAME: 'WORSHIP-SCHEDULE_CURRENT_SCORE'
});

function doPost(e) {
  const p = (e && e.parameter) || {};
  const requestId = String(p.requestId || '');
  try {
    if (String(p.app || '') !== SCORE_STORE.APP) {
      return scoreReply_({ ok: false, error: '지원하지 않는 요청입니다.' }, requestId);
    }

    const action = String(p.action || '');
    if (action === 'meta') {
      return scoreReply_({ ok: true, file: getCurrentScoreMeta_() }, requestId);
    }
    if (action === 'upload') {
      return scoreReply_(uploadCurrentScore_(p), requestId);
    }
    if (action === 'delete') {
      deleteCurrentScore_();
      return scoreReply_({ ok: true, file: null }, requestId);
    }

    return scoreReply_({ ok: false, error: '지원하지 않는 악보 작업입니다.' }, requestId);
  } catch (err) {
    console.error(err && err.stack ? err.stack : err);
    return scoreReply_({ ok: false, error: String(err && err.message ? err.message : err) }, requestId);
  }
}

function uploadCurrentScore_(p) {
  const fileName = sanitizeScoreFileName_(String(p.fileName || '').trim());
  const mimeType = normalizeScoreMime_(String(p.mimeType || ''), fileName);
  const base64 = String(p.dataBase64 || '');

  if (!fileName) throw new Error('파일명이 비어 있습니다.');
  if (!isAllowedScoreFile_(fileName, mimeType)) {
    throw new Error('PDF, PPT, PPTX, DOC, DOCX, JPG, JPEG, PNG 파일만 올릴 수 있습니다.');
  }
  if (!base64) throw new Error('파일 데이터가 비어 있습니다.');

  const bytes = Utilities.base64Decode(base64);
  const maxBytes = 25 * 1024 * 1024;
  if (bytes.length > maxBytes) throw new Error('파일은 25MB 이하만 업로드할 수 있습니다.');

  const folder = getScoreFolder_();
  const blob = Utilities.newBlob(bytes, mimeType, fileName);
  const newFile = folder.createFile(blob);

  try {
    newFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (sharingErr) {
    newFile.setTrashed(true);
    throw new Error('미리보기 공유 설정에 실패했습니다: ' + sharingErr.message);
  }

  // 새 파일 생성이 성공한 뒤 이전 파일을 제거합니다.
  // 업로드 실패 시 기존 악보까지 사라지는 상황을 방지합니다.
  const files = folder.getFiles();
  while (files.hasNext()) {
    const f = files.next();
    if (f.getId() !== newFile.getId()) {
      try { f.setTrashed(true); } catch (e) {}
    }
  }

  const props = PropertiesService.getScriptProperties();
  const updatedAt = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss');
  props.setProperties({
    [SCORE_STORE.FILE_PROP]: newFile.getId(),
    [SCORE_STORE.NAME_PROP]: fileName,
    [SCORE_STORE.MIME_PROP]: mimeType,
    [SCORE_STORE.UPDATED_PROP]: updatedAt
  }, true);

  return { ok: true, file: buildScoreMeta_(newFile, updatedAt) };
}

function deleteCurrentScore_() {
  const props = PropertiesService.getScriptProperties();
  const folder = getScoreFolder_();
  const files = folder.getFiles();
  while (files.hasNext()) {
    try { files.next().setTrashed(true); } catch (e) {}
  }
  props.deleteProperty(SCORE_STORE.FILE_PROP);
  props.deleteProperty(SCORE_STORE.NAME_PROP);
  props.deleteProperty(SCORE_STORE.MIME_PROP);
  props.deleteProperty(SCORE_STORE.UPDATED_PROP);
}

function getCurrentScoreMeta_() {
  const props = PropertiesService.getScriptProperties();
  const fileId = props.getProperty(SCORE_STORE.FILE_PROP) || '';
  if (!fileId) return null;

  try {
    const file = DriveApp.getFileById(fileId);
    if (file.isTrashed()) return null;
    return buildScoreMeta_(file, props.getProperty(SCORE_STORE.UPDATED_PROP) || '');
  } catch (e) {
    props.deleteProperty(SCORE_STORE.FILE_PROP);
    props.deleteProperty(SCORE_STORE.NAME_PROP);
    props.deleteProperty(SCORE_STORE.MIME_PROP);
    props.deleteProperty(SCORE_STORE.UPDATED_PROP);
    return null;
  }
}

function buildScoreMeta_(file, updatedAt) {
  const id = file.getId();
  return {
    id: id,
    name: file.getName(),
    mimeType: file.getMimeType(),
    size: file.getSize(),
    updatedAt: updatedAt || '',
    previewUrl: 'https://drive.google.com/file/d/' + encodeURIComponent(id) + '/preview',
    openUrl: 'https://drive.google.com/file/d/' + encodeURIComponent(id) + '/view'
  };
}

function getScoreFolder_() {
  const props = PropertiesService.getScriptProperties();
  const savedId = props.getProperty(SCORE_STORE.FOLDER_PROP) || '';
  if (savedId) {
    try {
      const folder = DriveApp.getFolderById(savedId);
      if (!folder.isTrashed()) return folder;
    } catch (e) {}
  }

  const folder = DriveApp.createFolder(SCORE_STORE.FOLDER_NAME);
  props.setProperty(SCORE_STORE.FOLDER_PROP, folder.getId());
  return folder;
}

function scoreReply_(data, requestId) {
  const payload = Object.assign({ source: 'worship-score-api', requestId: String(requestId || '') }, data || {});
  const json = JSON.stringify(payload).replace(/</g, '\\u003c');
  return HtmlService
    .createHtmlOutput('<!doctype html><meta charset="utf-8"><script>window.parent.postMessage(' + json + ', "*");</script>')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function sanitizeScoreFileName_(name) {
  return String(name || '').replace(/[\\/:*?"<>|]/g, '_').slice(0, 180);
}

function normalizeScoreMime_(mimeType, fileName) {
  const ext = String(fileName || '').toLowerCase().split('.').pop();
  const map = {
    pdf: 'application/pdf',
    ppt: 'application/vnd.ms-powerpoint',
    pptx: 'application/vnd.openxmlformats-officedocument.presentationml.presentation',
    doc: 'application/msword',
    docx: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    jpg: 'image/jpeg',
    jpeg: 'image/jpeg',
    png: 'image/png'
  };
  return map[ext] || mimeType || 'application/octet-stream';
}

function isAllowedScoreFile_(fileName, mimeType) {
  const ext = String(fileName || '').toLowerCase().split('.').pop();
  return ['pdf','ppt','pptx','doc','docx','jpg','jpeg','png'].includes(ext) && !!mimeType;
}
