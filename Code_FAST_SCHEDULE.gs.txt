 * - 기존 찬양팀 스프레드시트를 WORSHIP_SPREADSHEET_ID로 연결
 */

function doGet(e) { return worshipHandleRequest_(e); }

function doPost(e) {
  const p = (e && e.parameter) || {};
  const requestId = String(p.requestId || '');
  let result;
  try {
    if (String(p.app || '') !== 'worship') {
      result = { ok: false, error: 'worship app 요청이 아닙니다.' };
    } else {
      result = worshipDispatch_(p);
    }
  } catch (err) {
    console.error(err && err.stack ? err.stack : err);
    result = { ok: false, error: String(err && err.message ? err.message : err) };
  }
  return worshipBridgeOutput_(result, requestId);
}

function worshipBridgeOutput_(data, requestId) {
  const message = {
    source: 'worship-schedule-api',
    requestId: String(requestId || ''),
    payload: data
  };
  const safeJson = JSON.stringify(message)
    .replace(/</g, '\\u003c')
    .replace(/>/g, '\\u003e')
    .replace(/&/g, '\\u0026');
  const html = '<!doctype html><html><head><meta charset="utf-8"></head><body><script>' +
    '(function(){var m=' + safeJson + ';' +
    'try{window.top.postMessage(m,"*");}catch(e){}' +
    'try{window.parent.postMessage(m,"*");}catch(e){}' +
    '})();<\\/script></body></html>';
  return HtmlService.createHtmlOutput(html)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function worshipDispatch_(p) {
  const action = String((p && p.action) || '').trim();
  if (action === 'bootstrap') return worshipBootstrap_();
  if (action === 'revision') return { ok: true, revision: worshipGetRevision_() };
  if (action === 'mutate') return worshipMutate_(worshipParseJson_(p.payload, null));
  return { ok: false, error: '지원하지 않는 worship action: ' + action };
}

function worshipHandleRequest_(e) {
  const p = (e && e.parameter) || {};
  const callback = String(p.callback || '');
  try {
    return worshipOutput_(worshipDispatch_(p), callback);
  } catch (err) {
    console.error(err && err.stack ? err.stack : err);
    return worshipOutput_({ ok: false, error: String(err && err.message ? err.message : err) }, callback);
  }
}

function worshipOutput_(data, callback) {
  const json = JSON.stringify(data);
  if (callback) {
    const cb = String(callback).replace(/[^a-zA-Z0-9_$\.]/g, '');
    return ContentService.createTextOutput(cb + '(' + json + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}

function worshipBootstrap_() {
  const cache = CacheService.getScriptCache();
  try { const cached = cache.get(WORSHIP_API.CACHE_KEY); if (cached) return { ok: true, data: JSON.parse(cached) }; } catch (_) {}
  const ss = worshipDb_();
  const configRows = worshipReadTable_(ss, WORSHIP_API.SHEETS.CONFIG).objects;
  const config = {};
  configRows.forEach(r => { config[String(r.key || '')] = String(r.value == null ? '' : r.value); });
  const data = {
    revision: Number(config.STATE_REVISION || 1),
    serverToday: Utilities.formatDate(new Date(), WORSHIP_API.TZ, 'yyyy-MM-dd'),
    config,
    roles: worshipReadTable_(ss, WORSHIP_API.SHEETS.ROLES).objects.filter(r => worshipTruthy_(r.active)).sort((a,b)=>worshipNum_(a.display_order)-worshipNum_(b.display_order)),
    members: worshipReadTable_(ss, WORSHIP_API.SHEETS.MEMBERS).objects.filter(r => worshipTruthy_(r.active)).sort((a,b)=>worshipNum_(a.display_order)-worshipNum_(b.display_order)),
    memberRoles: worshipReadTable_(ss, WORSHIP_API.SHEETS.MEMBER_ROLES).objects.filter(r => worshipTruthy_(r.active)),
    events: worshipReadTable_(ss, WORSHIP_API.SHEETS.EVENTS).objects,
    assignments: worshipReadTable_(ss, WORSHIP_API.SHEETS.ASSIGNMENTS).objects.filter(r => worshipTruthy_(r.active)),
    prayerOrder: worshipReadTable_(ss, WORSHIP_API.SHEETS.PRAYER_ORDER).objects.filter(r => worshipTruthy_(r.active)).sort((a,b)=>worshipNum_(a.position)-worshipNum_(b.position)),
    rotationBase: worshipReadTable_(ss, WORSHIP_API.SHEETS.ROTATION_BASE).objects.filter(r => worshipTruthy_(r.active)),
    holidays: worshipReadTable_(ss, WORSHIP_API.SHEETS.HOLIDAYS).objects
  };
  try {
  cache.put(WORSHIP_API.CACHE_KEY, JSON.stringify(data), WORSHIP_API.CACHE_SECONDS);
  cache.put(WORSHIP_API.REV_CACHE_KEY, String(data.revision || 1), WORSHIP_API.REV_CACHE_SECONDS);
} catch (_) {}
return { ok: true, data };
}

function worshipGetRevision_() {
  const cache = CacheService.getScriptCache();
  try {
    const cached = cache.get(WORSHIP_API.REV_CACHE_KEY);
    if (cached !== null && cached !== '') return Number(cached || 1);
  } catch (_) {}

  const ss = worshipDb_();
  const sheet = ss.getSheetByName(WORSHIP_API.SHEETS.CONFIG);
  if (!sheet) throw new Error('필수 시트가 없습니다: ' + WORSHIP_API.SHEETS.CONFIG);
  const lastRow = sheet.getLastRow();
  let revision = 1;
  if (lastRow >= 2) {
    const rows = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
    for (let i = 0; i < rows.length; i++) {
      if (String(rows[i][0]) === 'STATE_REVISION') {
        revision = Number(rows[i][1] || 1);
        break;
      }
    }
  }
  try { cache.put(WORSHIP_API.REV_CACHE_KEY, String(revision), WORSHIP_API.REV_CACHE_SECONDS); } catch (_) {}
  return revision;
}

function worshipMutate_(mutation) {
  if (!mutation || typeof mutation !== 'object') throw new Error('저장 요청 형식이 올바르지 않습니다.');
  const mutationId = String(mutation.id || '');
  const clientId = String(mutation.clientId || '');
  const type = String(mutation.type || '');
  const payload = mutation.payload || {};
  if (!mutationId || !type) throw new Error('mutation id/type이 없습니다.');
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(1800)) return { ok:true, busy:true, retryAfterMs:900 };
  try {
    const ss = worshipDb_();
    if (worshipMutationAlreadyApplied_(ss, mutationId)) return { ok:true, revision:worshipGetRevision_(), duplicate:true };
    const now = worshipNow_();
    let structural = false;
    switch(type) {
      case 'assignmentSet': worshipSetAssignment_(ss,payload,now); break;
      case 'eventSet': worshipSetEvent_(ss,payload,now); break;
      case 'memberSave': structural=worshipSaveMember_(ss,payload,now); break;
      case 'memberDelete': structural=true; worshipDeleteMember_(ss,payload,now); break;
      case 'roleSave': structural=worshipSaveRole_(ss,payload,now); break;
      case 'roleDelete': structural=true; worshipDeleteRole_(ss,payload,now); break;
      case 'prayerOrderSet': structural=true; worshipSetPrayerOrder_(ss,payload,now); break;
      case 'rotationComplete': structural=true; worshipSaveRotationSnapshot_(ss,payload,now); break;
      default: throw new Error('지원하지 않는 저장 작업: ' + type);
    }
    const revision = worshipIncrementRevision_(ss, now);
    worshipAppendMutationLog_(ss, mutationId, clientId, type, now);
    SpreadsheetApp.flush();
    const scriptCache = CacheService.getScriptCache();
  scriptCache.remove(WORSHIP_API.CACHE_KEY);
  try { scriptCache.put(WORSHIP_API.REV_CACHE_KEY, String(revision), WORSHIP_API.REV_CACHE_SECONDS); } catch (_) {}
  return { ok:true, revision, structural };
  } catch(err) {
    console.error(err && err.stack ? err.stack : err);
    return { ok:false, error:String(err && err.message ? err.message : err) };
  } finally { lock.releaseLock(); }
}

function worshipSetAssignment_(ss,p,now) {
  const date=String(p.date||''), roleId=String(p.roleId||'');
  if(!/^\d{4}-\d{2}-\d{2}$/.test(date)||!roleId) throw new Error('배정 날짜/역할이 올바르지 않습니다.');
  const role=worshipReadTable_(ss,WORSHIP_API.SHEETS.ROLES).objects.find(r=>String(r.role_id)===roleId&&worshipTruthy_(r.active));
  if(!role) throw new Error('존재하지 않는 역할입니다.');
  let memberIds=Array.isArray(p.memberIds)?p.memberIds.map(String).filter(Boolean):[];
  memberIds=[...new Set(memberIds)]; if(!worshipTruthy_(role.allow_multiple)) memberIds=memberIds.slice(0,1);
  const table=worshipReadTable_(ss,WORSHIP_API.SHEETS.ASSIGNMENTS);
  table.objects.forEach(r=>{ if(worshipIsoDate_(r.event_date)===date&&String(r.role_id)===roleId&&worshipTruthy_(r.active)){r.active=false;r.updated_at=now;table.dirty=true;} });
  if(!memberIds.length){ table.objects.push({assignment_id:worshipId_('a'),event_date:date,role_id:roleId,member_id:'',sort_order:0,source:'manual_clear',active:true,updated_at:now}); table.dirty=true; }
  else memberIds.forEach((memberId,i)=>{table.objects.push({assignment_id:worshipId_('a'),event_date:date,role_id:roleId,member_id:memberId,sort_order:i+1,source:'manual',active:true,updated_at:now});table.dirty=true;});
  worshipWriteTable_(table);
}

function worshipSetEvent_(ss,p,now) {
  const date=String(p.date||''); if(!/^\d{4}-\d{2}-\d{2}$/.test(date)) throw new Error('일정 날짜가 올바르지 않습니다.');
  const table=worshipReadTable_(ss,WORSHIP_API.SHEETS.EVENTS); let row=table.objects.find(r=>worshipIsoDate_(r.event_date)===date);
  if(!row){row={event_date:date,event_type:'worship',title:String(p.title||''),active:!!p.active,source:'manual',updated_at:now};table.objects.push(row);} else {row.event_type=String(row.event_type||'worship');row.title=String(p.title==null?row.title||'':p.title);row.active=!!p.active;row.source='manual';row.updated_at=now;}
  table.dirty=true; worshipWriteTable_(table);
}

function worshipSaveMember_(ss,p,now) {
  const table=worshipReadTable_(ss,WORSHIP_API.SHEETS.MEMBERS), mr=worshipReadTable_(ss,WORSHIP_API.SHEETS.MEMBER_ROLES), name=String(p.name||'').trim();
  if(!name) throw new Error('멤버 이름을 입력해주세요.');
  const isNew=!String(p.memberId||''), memberId=isNew?worshipId_('m'):String(p.memberId); let row=table.objects.find(r=>String(r.member_id)===memberId);
  if(!row){const max=table.objects.reduce((m,r)=>Math.max(m,worshipNum_(r.display_order)),0);row={member_id:memberId,member_name:name,display_order:max+1,active:true,updated_at:now};table.objects.push(row);} else {row.member_name=name;row.active=true;row.updated_at=now;}
  table.dirty=true;
  mr.objects.forEach(r=>{if(String(r.member_id)===memberId&&worshipTruthy_(r.active)){r.active=false;r.updated_at=now;mr.dirty=true;}});
  const roleIds=Array.isArray(p.roleIds)?[...new Set(p.roleIds.map(String).filter(Boolean))]:[];
  roleIds.forEach(roleId=>{let link=mr.objects.find(r=>String(r.member_id)===memberId&&String(r.role_id)===roleId);if(!link){link={member_id:memberId,role_id:roleId,active:true,updated_at:now};mr.objects.push(link);}else{link.active=true;link.updated_at=now;}mr.dirty=true;});
  worshipWriteTable_(table); worshipWriteTable_(mr); return isNew;
}

function worshipDeleteMember_(ss,p,now) {
  const id=String(p.memberId||''); if(!id) throw new Error('삭제할 멤버 ID가 없습니다.');
  const table=worshipReadTable_(ss,WORSHIP_API.SHEETS.MEMBERS), row=table.objects.find(r=>String(r.member_id)===id);
  if(row){row.active=false;row.updated_at=now;table.dirty=true;worshipWriteTable_(table);}
  const mr=worshipReadTable_(ss,WORSHIP_API.SHEETS.MEMBER_ROLES); mr.objects.forEach(r=>{if(String(r.member_id)===id&&worshipTruthy_(r.active)){r.active=false;r.updated_at=now;mr.dirty=true;}}); if(mr.dirty) worshipWriteTable_(mr);
  const prayer=worshipReadTable_(ss,WORSHIP_API.SHEETS.PRAYER_ORDER), before=prayer.objects.filter(r=>worshipTruthy_(r.active)), active=before.filter(r=>String(r.member_id)!==id);
  if(active.length!==before.length){prayer.objects=active.map((r,i)=>({position:i+1,member_id:String(r.member_id||''),display_name:String(r.display_name||''),active:true,updated_at:now}));prayer.dirty=true;worshipWriteTable_(prayer);}
}

function worshipSetPrayerOrder_(ss,p,now) {
  const ids=Array.isArray(p.memberIds)?p.memberIds.map(String).filter(Boolean):[], members=worshipReadTable_(ss,WORSHIP_API.SHEETS.MEMBERS).objects, names={}; members.forEach(r=>{names[String(r.member_id)]=String(r.member_name||'');});
  const table=worshipReadTable_(ss,WORSHIP_API.SHEETS.PRAYER_ORDER); table.objects=ids.map((id,i)=>({position:i+1,member_id:id,display_name:names[id]||'',active:true,updated_at:now})); table.dirty=true; worshipWriteTable_(table);
}

function worshipSaveRotationSnapshot_(ss,p,now) {
  const month=String(p.month||''); if(!/^\d{4}-\d{2}$/.test(month)) throw new Error('편성 기준 월이 올바르지 않습니다.');
  const events=worshipReadTable_(ss,WORSHIP_API.SHEETS.EVENTS).objects;
  const assignments=worshipReadTable_(ss,WORSHIP_API.SHEETS.ASSIGNMENTS).objects.filter(r=>worshipTruthy_(r.active));
  const rotationTable=worshipReadTable_(ss,WORSHIP_API.SHEETS.ROTATION_BASE), activeRotation=rotationTable.objects.filter(r=>worshipTruthy_(r.active));
  const roles=worshipReadTable_(ss,WORSHIP_API.SHEETS.ROLES).objects.filter(r=>worshipTruthy_(r.active)), dates=worshipSelectedDates_(month,events);
  if(!dates.length) throw new Error('저장할 찬양 일정이 없습니다.');
  const inherited=worshipLatestBaseBefore_(month,activeRotation), snapshot=dates.map(date=>{const slot={};roles.forEach(role=>{slot[String(role.role_id)]=worshipEffectiveMemberIds_(date,String(role.role_id),month,dates,assignments,activeRotation,inherited);});return slot;});
  rotationTable.objects.forEach(r=>{if(String(r.base_month)===month&&worshipTruthy_(r.active)){r.active=false;r.updated_at=now;rotationTable.dirty=true;}});
  snapshot.forEach((slot,index)=>Object.keys(slot).forEach(roleId=>(slot[roleId]||[]).forEach((memberId,order)=>{rotationTable.objects.push({base_month:month,event_index:index+1,role_id:roleId,member_id:String(memberId),sort_order:order+1,active:true,updated_at:now});rotationTable.dirty=true;})));
  worshipWriteTable_(rotationTable);
}

function worshipSelectedDates_(month,events) {
  const parts=month.split('-').map(Number), y=parts[0], m=parts[1], map={}, last=new Date(y,m,0).getDate();
  for(let d=1;d<=last;d++){const dt=new Date(y,m-1,d);if(dt.getDay()===0)map[y+'-'+('0'+m).slice(-2)+'-'+('0'+d).slice(-2)]=true;}
  (events||[]).forEach(r=>{const date=worshipIsoDate_(r.event_date);if(date.slice(0,7)===month)map[date]=worshipTruthy_(r.active);});
  return Object.keys(map).filter(k=>map[k]).sort();
}
function worshipLatestBaseBefore_(month,rows) { const target=worshipMonthNum_(month), months=[...new Set((rows||[]).filter(r=>worshipTruthy_(r.active)).map(r=>String(r.base_month||'')).filter(x=>/^\d{4}-\d{2}$/.test(x)&&worshipMonthNum_(x) <= target))].sort((a,b)=>worshipMonthNum_(a)-worshipMonthNum_(b)); return months.length?months[months.length-1]:''; }
function worshipEffectiveMemberIds_(date,roleId,month,monthDates,assignments,rotation,inheritedBase) { const explicit=(assignments||[]).filter(a=>worshipTruthy_(a.active)&&worshipIsoDate_(a.event_date)===date&&String(a.role_id)===roleId).sort((a,b)=>worshipNum_(a.sort_order)-worshipNum_(b.sort_order)); if(explicit.length)return explicit.filter(a=>String(a.member_id||'')).map(a=>String(a.member_id)); if(!inheritedBase)return[]; const targetIndex=monthDates.indexOf(date); if(targetIndex<0)return[]; const baseRows=(rotation||[]).filter(r=>worshipTruthy_(r.active)&&String(r.base_month)===inheritedBase), indices=[...new Set(baseRows.map(r=>worshipNum_(r.event_index)).filter(Boolean))].sort((a,b)=>a-b); if(!indices.length)return[]; const patternIndex=indices[targetIndex%indices.length]; return baseRows.filter(r=>worshipNum_(r.event_index)===patternIndex&&String(r.role_id)===roleId&&String(r.member_id||'')).sort((a,b)=>worshipNum_(a.sort_order)-worshipNum_(b.sort_order)).map(r=>String(r.member_id)); }

function worshipMutationAlreadyApplied_(ss,mutationId) { const sheet=ss.getSheetByName(WORSHIP_API.SHEETS.MUTATION_LOG); if(!sheet||sheet.getLastRow()<2)return false; const count=Math.min(Math.max(sheet.getLastRow()-1,0),WORSHIP_API.LOG_KEEP); if(!count)return false; const start=sheet.getLastRow()-count+1, values=sheet.getRange(start,1,count,1).getDisplayValues(); return values.some(r=>String(r[0])===mutationId); }
function worshipAppendMutationLog_(ss,mutationId,clientId,type,now) { const sheet=ss.getSheetByName(WORSHIP_API.SHEETS.MUTATION_LOG); if(!sheet)throw new Error('필수 시트가 없습니다: '+WORSHIP_API.SHEETS.MUTATION_LOG); sheet.appendRow([mutationId,clientId,type,now]); const rows=Math.max(sheet.getLastRow()-1,0); if(rows>WORSHIP_API.LOG_KEEP)sheet.deleteRows(2,rows-WORSHIP_API.LOG_KEEP); }
function worshipIncrementRevision_(ss,now) { const sheet=ss.getSheetByName(WORSHIP_API.SHEETS.CONFIG); if(!sheet)throw new Error('필수 시트가 없습니다: '+WORSHIP_API.SHEETS.CONFIG); const last=sheet.getLastRow(); if(last>=2){const values=sheet.getRange(2,1,last-1,2).getValues();for(let i=0;i<values.length;i++){if(String(values[i][0])==='STATE_REVISION'){const next=Number(values[i][1]||0)+1;sheet.getRange(i+2,2).setValue(String(next));return next;}}} sheet.appendRow(['STATE_REVISION','1','공용 데이터 변경 버전']); return 1; }

function setupWorshipStandalone() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('찬양팀_DB 스프레드시트에서 확장 프로그램 > Apps Script로 연 뒤 실행해주세요.');
  PropertiesService.getScriptProperties().setProperty(WORSHIP_API.DB_PROP, ss.getId());
  worshipValidateSchema_(ss);
  // Force the exact Sheets permission the deployed web app will use.
  const reopened = SpreadsheetApp.openById(ss.getId());
  reopened.getName();
  SpreadsheetApp.flush();
  return '일정 백엔드 연결/권한 확인 완료: ' + ss.getName();
}

function setWorshipSpreadsheetId(id) {
  const value = String(id || '').trim();
  if (!value) throw new Error('스프레드시트 ID가 비어 있습니다.');
  const ss = SpreadsheetApp.openById(value);
  PropertiesService.getScriptProperties().setProperty(WORSHIP_API.DB_PROP, ss.getId());
  return ss.getName();
}

function worshipDb_() {
  const id = PropertiesService.getScriptProperties().getProperty(WORSHIP_API.DB_PROP);
  if (!id) throw new Error('WORSHIP_SPREADSHEET_ID가 없습니다. setWorshipSpreadsheetId("스프레드시트ID")를 한 번 실행해주세요.');
  return SpreadsheetApp.openById(id);
}
function worshipValidateSchema_(ss) { Object.keys(WORSHIP_API.HEADERS).forEach(sheetName=>{const sheet=ss.getSheetByName(sheetName);if(!sheet)throw new Error('필수 시트가 없습니다: '+sheetName);const expected=WORSHIP_API.HEADERS[sheetName], actual=sheet.getRange(1,1,1,expected.length).getDisplayValues()[0].map(String);expected.forEach((h,i)=>{if(actual[i]!==h)throw new Error(sheetName+' 시트 '+(i+1)+'열은 ['+h+'] 이어야 합니다.');});}); }
function worshipReadTable_(ss,sheetName) { const sheet=ss.getSheetByName(sheetName);if(!sheet)throw new Error('시트 없음: '+sheetName);const headers=WORSHIP_API.HEADERS[sheetName], last=Math.max(sheet.getLastRow(),1), values=last>1?sheet.getRange(2,1,last-1,headers.length).getValues():[], objects=values.filter(row=>row.some(v=>v!==''&&v!=null)).map(row=>{const o={};headers.forEach((h,i)=>{o[h]=row[i];});return o;});return{sheet,headers,objects,oldCount:Math.max(last-1,0),dirty:false}; }
function worshipWriteTable_(table) { if(!table||!table.dirty)return;const rows=table.objects.map(o=>table.headers.map(h=>o[h]==null?'':o[h])), clearCount=Math.max(table.oldCount,rows.length);if(clearCount>0)table.sheet.getRange(2,1,clearCount,table.headers.length).clearContent();if(rows.length)table.sheet.getRange(2,1,rows.length,table.headers.length).setValues(rows);table.oldCount=rows.length;table.dirty=false; }
function worshipParseJson_(value,fallback){if(value===''||value==null)return fallback;try{return JSON.parse(String(value));}catch(_){return fallback;}}
function worshipTruthy_(v){if(v===true)return true;const s=String(v==null?'':v).trim().toLowerCase();return s==='true'||s==='1'||s==='y'||s==='yes';}
function worshipNum_(v){const n=Number(v);return Number.isFinite(n)?n:0;}
function worshipIsoDate_(v){if(!v&&v!==0)return'';if(Object.prototype.toString.call(v)==='[object Date]'&&!isNaN(v))return Utilities.formatDate(v,WORSHIP_API.TZ,'yyyy-MM-dd');const s=String(v).trim(),m=s.match(/^(\d{4})[-\/.](\d{1,2})[-\/.](\d{1,2})/);if(!m)return s;return m[1]+'-'+('0'+m[2]).slice(-2)+'-'+('0'+m[3]).slice(-2);}
function worshipMonthNum_(month){const p=String(month||'').split('-').map(Number);return p[0]*12+(p[1]-1);}
