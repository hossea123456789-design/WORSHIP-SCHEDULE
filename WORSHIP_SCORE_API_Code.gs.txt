/**
 * WORSHIP_SCORE_API v3.12.9
 * 찬양팀 악보 전용 standalone Apps Script backend
 *
 * 현재 프런트(v3.12.6) 계약:
 * - POST app=worship-score
 * - action=health | meta | uploadDirect | uploadStart | uploadChunk | delete
 * - 모든 POST 응답은 iframe postMessage bridge로 requestId를 그대로 돌려줌
 * - 4MB 이하: uploadDirect
 * - 4MB 초과~25MB: Google Drive resumable upload
 * - 새 파일이 정상 저장/공유된 뒤에만 이전 악보를 휴지통으로 이동
 * - 업로드 응답 유실 시 meta/세션 재호출로 복구 가능
 *
 * 필수 OAuth scope:
 * - https://www.googleapis.com/auth/drive
 * - https://www.googleapis.com/auth/script.external_request
 */

const SCORE_API_VERSION = 'v3.12.9';

const SCORE_REQUIRED_SCOPES = Object.freeze([
  'https://www.googleapis.com/auth/drive',
  'https://www.googleapis.com/auth/script.external_request'
]);

const SCORE_STORE = Object.freeze({
  APP: 'worship-score',
  FOLDER_PROP: 'WORSHIP_SCORE_FOLDER_ID',
  CANONICAL_FOLDER_ID: '1qb-KEOyDzXbwD0-18gUFP_rfy7zNcW-D',
  FILE_PROP: 'WORSHIP_SCORE_FILE_ID',
  NAME_PROP: 'WORSHIP_SCORE_FILE_NAME',
  MIME_PROP: 'WORSHIP_SCORE_FILE_MIME',
  UPDATED_PROP: 'WORSHIP_SCORE_UPDATED_AT',
  UPLOAD_PREFIX: 'WORSHIP_SCORE_UPLOAD_',
  FOLDER_NAME: 'WORSHIP-SCHEDULE_CURRENT_SCORE',
  CHUNK_SIZE: 768 * 1024,
  DIRECT_MAX_BYTES: 4 * 1024 * 1024,
  MAX_BYTES: 25 * 1024 * 1024,
  STALE_UPLOAD_MS: 24 * 60 * 60 * 1000
});

const AFTER_SCORE_STORE = Object.freeze({
  FOLDER_PROP: 'WORSHIP_AFTER_SCORE_FOLDER_ID',
  FILE_PROP: 'WORSHIP_AFTER_SCORE_FILE_ID',

  FOLDER_NAME: 'WORSHIP-SCHEDULE_AFTER_SERVICE_DATA',
  FILE_NAME: 'youtube-links.json'
});

function resetScoreAuthorization() {
  ScriptApp.invalidateAuth();
  return '기존 Apps Script 권한을 초기화했습니다. 편집기를 새로고침한 뒤 setupScoreStandalone을 실행해주세요. 누락된 권한은 setupScoreStandalone이 자동으로 요청합니다.';
}

function repairScoreFolderReference() {
  const folder = getScoreFolder_(true);
  return [
    '악보 폴더 참조 복구 완료',
    '폴더명: ' + folder.getName(),
    '폴더 ID: ' + folder.getId(),
    '새 폴더 생성 여부: 기존 폴더 재사용'
  ].join('\n');
}

function setupScoreStandalone() {
  // Apps Script IDE에서 실행하면 누락된 세부 OAuth 동의를 자동으로 요청합니다.
  ScriptApp.requireScopes(ScriptApp.AuthMode.FULL, SCORE_REQUIRED_SCOPES);

  const folder = getScoreFolder_(true);
  const afterFolder = getAfterScoreFolder_(true);
  const afterLinkFolder = getAfterLinkFolder_(true);
  const rootName = DriveApp.getRootFolder().getName();

  // DriveApp 자체의 쓰기 권한을 실제로 확인합니다.
  const currentDescription = folder.getDescription() || '';
  folder.setDescription(currentDescription);

  // UrlFetch 권한을 확인합니다.
  const probe = UrlFetchApp.fetch('https://www.google.com/generate_204', {
    method: 'get',
    muteHttpExceptions: true,
    followRedirects: true
  });
  const status = probe.getResponseCode();
  if (status < 200 || status >= 400) {
    throw new Error('외부 요청 권한 확인 실패: HTTP ' + status);
  }

  // ScriptApp.getOAuthToken()이 Drive API "쓰기" scope까지 실제로 갖는지 확인합니다.
  verifyDriveApiWriteScope_(folder);

  return [
    'WORSHIP_SCORE_API 설정 완료',
    '버전: ' + SCORE_API_VERSION,
    'Drive 루트: ' + rootName,
    '주일 악보 폴더: ' + folder.getName(),
    '주일 악보 폴더 ID: ' + folder.getId(),
    '예배 후 악보 폴더: ' + afterFolder.getName(),
    '예배 후 악보 폴더 ID: ' + afterFolder.getId(),
    '예배 후 링크 데이터 폴더: ' + afterLinkFolder.getName(),
    'Drive API 쓰기 권한: 정상'
  ].join('\n');
}

function doGet(e) {
  const p = (e && e.parameter) || {};
  const action = String(p.action || 'health').trim();
  const callback = String(p.callback || '').trim();

  try {
    let data;
    if (action === 'health') {
      data = scoreHealth_();
    } else if (action === 'meta') {
      data = {
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        file: getCurrentScoreMeta_()
      };
    } else if (action === 'afterMeta') {
      data = {
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        file: getAfterScoreMeta_()
      };
    } else if (action === 'afterLinks') {
      data = {
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        links: getAfterYoutubeLinks_()
      };
    } else {
      data = {
        ok: false,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        error: '지원하지 않는 GET action: ' + action
      };
    }
    return scoreTextOutput_(data, callback);
  } catch (err) {
    return scoreTextOutput_({
      ok: false,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      error: errorMessage_(err)
    }, callback);
  }
}

function doPost(e) {
  const p = (e && e.parameter) || {};
  const requestId = String(p.requestId || '');

  try {
    if (String(p.app || '') !== SCORE_STORE.APP) {
      return scoreReply_({
        ok: false,
        error: '지원하지 않는 요청입니다.'
      }, requestId);
    }

    const action = String(p.action || '').trim();

    if (action === 'health') return scoreReply_(scoreHealth_(), requestId);

    if (action === 'meta') {
      return scoreReply_({
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        file: getCurrentScoreMeta_()
      }, requestId);
    }

    if (action === 'afterMeta') {
      return scoreReply_({
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        file: getAfterScoreMeta_()
      }, requestId);
    }
    if (action === 'afterLinks') {
      return scoreReply_({
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        links: getAfterYoutubeLinks_()
      }, requestId);
    }
    if (action === 'afterLinksSet') {
      return scoreReply_({
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        links: setAfterYoutubeLinks_(p.linksJson)
      }, requestId);
    }
    if (action === 'afterUploadDirect') {
      return scoreReply_(uploadAfterScoreDirect_(p), requestId);
    }
    if (action === 'afterUploadStart') {
      return scoreReply_(startAfterScoreUpload_(p), requestId);
    }
    if (action === 'afterUploadChunk') {
      return scoreReply_(uploadAfterScoreChunk_(p), requestId);
    }
    if (action === 'afterDelete') {
      deleteAfterScore_();
      return scoreReply_({
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        file: null
      }, requestId);
    }

    if (action === 'uploadDirect') {
      return scoreReply_(uploadScoreDirect_(p), requestId);
    }
    if (action === 'uploadStart') {
      return scoreReply_(startScoreUpload_(p), requestId);
    }
    if (action === 'uploadChunk') {
      return scoreReply_(uploadScoreChunk_(p), requestId);
    }
    if (action === 'delete') {
      deleteCurrentScore_();
      return scoreReply_({
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        file: null
      }, requestId);
    }

    return scoreReply_({
      ok: false,
      error: '지원하지 않는 악보 작업입니다: ' + action
    }, requestId);
  } catch (err) {
    console.error(err && err.stack ? err.stack : err);
    return scoreReply_({
      ok: false,
      error: errorMessage_(err)
    }, requestId);
  }
}

function scoreHealth_() {
  const folder = getScoreFolder_(true);
  const afterLinkFolder = getAfterLinkFolder_(true);
  DriveApp.getRootFolder().getName();

  const probe = UrlFetchApp.fetch('https://www.google.com/generate_204', {
    method: 'get',
    muteHttpExceptions: true,
    followRedirects: true
  });
  const status = probe.getResponseCode();
  if (status < 200 || status >= 400) {
    throw new Error('외부 요청 권한 확인 실패: HTTP ' + status);
  }

  // 업로드를 시작하기 전에 OAuth 토큰의 실제 Drive API 쓰기 scope를 검증합니다.
  verifyDriveApiWriteScope_(folder);

  return {
    ok: true,
    service: SCORE_STORE.APP,
    version: SCORE_API_VERSION,
    storageReady: true,
    afterLinksStorage: 'drive+properties',
    capabilities: ['meta','uploadDirect','uploadStart','uploadChunk','delete','afterMeta','afterUploadDirect','afterUploadStart','afterUploadChunk','afterDelete','afterLinks','afterLinksSet'],
    folderId: folder.getId(),
    afterLinkFolderId: afterLinkFolder.getId(),
    file: getCurrentScoreMeta_()
  };
}

function verifyDriveApiWriteScope_(folder) {
  const folderId = folder.getId();
  const url =
    'https://www.googleapis.com/drive/v3/files/' +
    encodeURIComponent(folderId) +
    '?fields=id,appProperties';

  const response = UrlFetchApp.fetch(url, {
    method: 'patch',
    contentType: 'application/json; charset=UTF-8',
    headers: {
      Authorization: 'Bearer ' + ScriptApp.getOAuthToken()
    },
    payload: JSON.stringify({
      appProperties: {
        worshipScoreApiProbe: 'ok'
      }
    }),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  if (status >= 200 && status < 300) return true;

  const body = truncate_(response.getContentText(), 700);
  if (status === 401 || status === 403) {
    throw new Error(
      'Google Drive 전체 쓰기 권한이 현재 OAuth 토큰에 없습니다. ' +
      'Apps Script에서 resetScoreAuthorization 실행 → 편집기 새로고침 → ' +
      'setupScoreStandalone 실행 후 Google Drive 접근 권한을 모두 허용해주세요. ' +
      '상세: ' + body
    );
  }

  throw new Error(
    'Drive API 권한 확인 실패 (' + status + '): ' + body
  );
}

function uploadScoreDirect_(p) {
  cleanupStaleScoreUploads_();

  const fileName = sanitizeScoreFileName_(String(p.fileName || '').trim());
  const mimeType = normalizeScoreMime_(String(p.mimeType || ''), fileName);
  const totalSize = Number(p.totalSize || 0);
  const dataBase64 = String(p.dataBase64 || '');

  validateScoreUpload_(fileName, mimeType, totalSize);

  if (totalSize > SCORE_STORE.DIRECT_MAX_BYTES) {
    throw new Error('직접 업로드 허용 크기를 초과했습니다. 분할 업로드를 사용해주세요.');
  }
  if (!dataBase64) throw new Error('파일 데이터가 비어 있습니다.');

  const bytes = Utilities.base64Decode(dataBase64);
  if (bytes.length !== totalSize) {
    throw new Error('파일 크기 정보가 일치하지 않습니다.');
  }

  const folder = getScoreFolder_(true);
  const blob = Utilities.newBlob(bytes, mimeType, fileName);
  const file = folder.createFile(blob);

  try {
    makeScoreFilePublic_(file);
    const updatedAt = finalizeCurrentScore_(file, fileName, mimeType);
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: true,
      transport: 'direct',
      receivedBytes: totalSize,
      file: buildScoreMeta_(file, updatedAt)
    };
  } catch (err) {
    try { file.setTrashed(true); } catch (_) {}
    throw err;
  }
}

function startScoreUpload_(p) {
  cleanupStaleScoreUploads_();

  const fileName = sanitizeScoreFileName_(String(p.fileName || '').trim());
  const mimeType = normalizeScoreMime_(String(p.mimeType || ''), fileName);
  const totalSize = Number(p.totalSize || 0);

  validateScoreUpload_(fileName, mimeType, totalSize);

  const folder = getScoreFolder_(true);
  const initUrl =
    'https://www.googleapis.com/upload/drive/v3/files' +
    '?uploadType=resumable&fields=id,name,mimeType,size';

  const response = UrlFetchApp.fetch(initUrl, {
    method: 'post',
    contentType: 'application/json; charset=UTF-8',
    headers: {
      Authorization: 'Bearer ' + ScriptApp.getOAuthToken(),
      'X-Upload-Content-Type': mimeType,
      'X-Upload-Content-Length': String(totalSize)
    },
    payload: JSON.stringify({
      name: fileName,
      parents: [folder.getId()]
    }),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  if (status < 200 || status >= 300) {
    const detail = truncate_(response.getContentText(), 700);
    if (status === 401 || status === 403) {
      throw new Error(
        'Google Drive API 쓰기 권한이 부족합니다. ' +
        'resetScoreAuthorization 실행 → 편집기 새로고침 → setupScoreStandalone 실행 후 ' +
        'Google Drive 접근 권한을 모두 허용해주세요. 상세: ' + detail
      );
    }
    throw new Error(
      'Drive 업로드 준비 실패 (' + status + '): ' + detail
    );
  }

  const sessionUrl = String(
    getHeaderIgnoreCase_(response.getAllHeaders(), 'Location') || ''
  );
  if (!sessionUrl) throw new Error('Drive 업로드 세션 URL을 받지 못했습니다.');

  const uploadId = Utilities.getUuid();
  const meta = {
    uploadId: uploadId,
    sessionUrl: sessionUrl,
    fileName: fileName,
    mimeType: mimeType,
    totalSize: totalSize,
    nextStart: 0,
    createdAt: Date.now(),
    complete: false,
    fileId: '',
    updatedAt: ''
  };

  PropertiesService.getScriptProperties().setProperty(
    SCORE_STORE.UPLOAD_PREFIX + uploadId,
    JSON.stringify(meta)
  );

  return {
    ok: true,
    service: SCORE_STORE.APP,
    version: SCORE_API_VERSION,
    transport: 'resumable',
    uploadId: uploadId,
    chunkSize: SCORE_STORE.CHUNK_SIZE,
    totalSize: totalSize
  };
}

function uploadScoreChunk_(p) {
  const uploadId = String(p.uploadId || '');
  if (!uploadId) throw new Error('업로드 세션 정보가 없습니다.');

  const props = PropertiesService.getScriptProperties();
  const key = SCORE_STORE.UPLOAD_PREFIX + uploadId;
  const raw = props.getProperty(key);
  if (!raw) {
    throw new Error('업로드 세션이 만료되었습니다. 처음부터 다시 업로드해주세요.');
  }

  let meta;
  try {
    meta = JSON.parse(raw);
  } catch (_) {
    props.deleteProperty(key);
    throw new Error('업로드 세션 정보가 손상되었습니다. 다시 업로드해주세요.');
  }

  if (meta.complete && meta.fileId) {
    const completedFile = DriveApp.getFileById(String(meta.fileId));
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: true,
      transport: 'resumable',
      receivedBytes: Number(meta.totalSize || 0),
      file: buildScoreMeta_(completedFile, String(meta.updatedAt || ''))
    };
  }

  const start = Number(p.start || 0);
  const end = Number(p.end || -1);
  const totalSize = Number(p.totalSize || meta.totalSize || 0);
  const dataBase64 = String(p.dataBase64 || '');

  if (!dataBase64) throw new Error('업로드 조각 데이터가 비어 있습니다.');
  if (totalSize !== Number(meta.totalSize)) {
    throw new Error('파일 크기 정보가 일치하지 않습니다.');
  }
  if (start < 0 || end < start || end >= totalSize) {
    throw new Error('업로드 범위가 올바르지 않습니다.');
  }

  const expectedNext = Number(meta.nextStart || 0);

  if (end < expectedNext) {
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: false,
      transport: 'resumable',
      receivedBytes: expectedNext
    };
  }

  if (start !== expectedNext) {
    throw new Error(
      '업로드 순서가 어긋났습니다. 예상 시작 위치: ' +
      expectedNext + ', 받은 위치: ' + start
    );
  }

  const bytes = Utilities.base64Decode(dataBase64);
  const expectedLength = end - start + 1;
  if (bytes.length !== expectedLength) {
    throw new Error('업로드 조각 크기가 일치하지 않습니다.');
  }

  const response = UrlFetchApp.fetch(String(meta.sessionUrl), {
    method: 'put',
    contentType: String(meta.mimeType || 'application/octet-stream'),
    headers: {
      'Content-Range': 'bytes ' + start + '-' + end + '/' + totalSize
    },
    payload: bytes,
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();

  if (status === 308) {
    meta.nextStart = end + 1;
    props.setProperty(key, JSON.stringify(meta));
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: false,
      transport: 'resumable',
      receivedBytes: meta.nextStart
    };
  }

  if (status >= 200 && status < 300) {
    let driveMeta = {};
    try { driveMeta = JSON.parse(response.getContentText() || '{}'); } catch (_) {}

    const fileId = String(driveMeta.id || '');
    if (!fileId) {
      throw new Error('Drive 업로드는 완료됐지만 파일 ID를 받지 못했습니다.');
    }

    const file = DriveApp.getFileById(fileId);

    try {
      makeScoreFilePublic_(file);
      const updatedAt = finalizeCurrentScore_(
        file,
        String(meta.fileName || file.getName()),
        String(meta.mimeType || file.getMimeType())
      );

      meta.complete = true;
      meta.fileId = fileId;
      meta.updatedAt = updatedAt;
      meta.nextStart = totalSize;
      props.setProperty(key, JSON.stringify(meta));

      return {
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        complete: true,
        transport: 'resumable',
        receivedBytes: totalSize,
        file: buildScoreMeta_(file, updatedAt)
      };
    } catch (err) {
      try { file.setTrashed(true); } catch (_) {}
      throw err;
    }
  }

  throw new Error(
    'Drive 업로드 실패 (' + status + '): ' +
    truncate_(response.getContentText(), 500)
  );
}

function validateScoreUpload_(fileName, mimeType, totalSize) {
  if (!fileName) throw new Error('파일명이 비어 있습니다.');
  if (!isAllowedScoreFile_(fileName, mimeType)) {
    throw new Error('PDF, PPT, PPTX, DOC, DOCX, JPG, JPEG, PNG 파일만 올릴 수 있습니다.');
  }
  if (!Number.isFinite(totalSize) || totalSize <= 0) {
    throw new Error('파일 크기를 확인할 수 없습니다.');
  }
  if (totalSize > SCORE_STORE.MAX_BYTES) {
    throw new Error('파일은 25MB 이하만 업로드할 수 있습니다.');
  }
}

function makeScoreFilePublic_(file) {
  try {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (err) {
    throw new Error('악보 미리보기 공유 설정에 실패했습니다: ' + errorMessage_(err));
  }
}

function finalizeCurrentScore_(newFile, fileName, mimeType) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const props = PropertiesService.getScriptProperties();
    const folder = getScoreFolder_(true);
    const updatedAt = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone() || 'Asia/Seoul',
      'yyyy-MM-dd HH:mm:ss'
    );

    props.setProperties({
      [SCORE_STORE.FILE_PROP]: newFile.getId(),
      [SCORE_STORE.NAME_PROP]: fileName || newFile.getName(),
      [SCORE_STORE.MIME_PROP]: mimeType || newFile.getMimeType(),
      [SCORE_STORE.UPDATED_PROP]: updatedAt
    }, true);

    const files = folder.getFiles();
    while (files.hasNext()) {
      const oldFile = files.next();
      if (oldFile.getId() !== newFile.getId()) {
        try { oldFile.setTrashed(true); } catch (_) {}
      }
    }

    return updatedAt;
  } finally {
    lock.releaseLock();
  }
}

function deleteCurrentScore_() {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const folder = getScoreFolder_(false);
    if (folder) {
      const files = folder.getFiles();
      while (files.hasNext()) {
        try { files.next().setTrashed(true); } catch (_) {}
      }
    }
    clearCurrentScoreProps_();
  } finally {
    lock.releaseLock();
  }
}

function getCurrentScoreMeta_() {
  const props = PropertiesService.getScriptProperties();
  const fileId = String(props.getProperty(SCORE_STORE.FILE_PROP) || '');

  if (fileId) {
    try {
      const file = DriveApp.getFileById(fileId);
      if (!file.isTrashed()) {
        return buildScoreMeta_(
          file,
          String(props.getProperty(SCORE_STORE.UPDATED_PROP) || '')
        );
      }
    } catch (_) {}
    clearCurrentScoreProps_();
  }

  const folder = getScoreFolder_(false);
  if (!folder) return null;

  const files = folder.getFiles();
  let newest = null;
  let newestTime = -1;

  while (files.hasNext()) {
    const file = files.next();
    try {
      if (file.isTrashed()) continue;
      const t = file.getLastUpdated().getTime();
      if (t > newestTime) {
        newest = file;
        newestTime = t;
      }
    } catch (_) {}
  }

  if (!newest) return null;

  const updatedAt = Utilities.formatDate(
    newest.getLastUpdated(),
    Session.getScriptTimeZone() || 'Asia/Seoul',
    'yyyy-MM-dd HH:mm:ss'
  );

  props.setProperties({
    [SCORE_STORE.FILE_PROP]: newest.getId(),
    [SCORE_STORE.NAME_PROP]: newest.getName(),
    [SCORE_STORE.MIME_PROP]: newest.getMimeType(),
    [SCORE_STORE.UPDATED_PROP]: updatedAt
  }, true);

  return buildScoreMeta_(newest, updatedAt);
}

function scoreFolderByIdIfUsable_(id) {
  const value = String(id || '').trim();
  if (!value) return null;
  try {
    const folder = DriveApp.getFolderById(value);
    if (!folder.isTrashed()) return folder;
  } catch (_) {}
  return null;
}

function folderContainsFileId_(folder, fileId) {
  const target = String(fileId || '').trim();
  if (!folder || !target) return false;
  try {
    const files = folder.getFiles();
    while (files.hasNext()) {
      if (String(files.next().getId()) === target) return true;
    }
  } catch (_) {}
  return false;
}

function findExistingScoreFolder_() {
  const props = PropertiesService.getScriptProperties();
  const currentFileId = String(props.getProperty(SCORE_STORE.FILE_PROP) || '');
  const root = DriveApp.getRootFolder();
  const it = root.getFoldersByName(SCORE_STORE.FOLDER_NAME);
  let newest = null;
  let newestTime = -1;

  while (it.hasNext()) {
    const folder = it.next();
    try {
      if (folder.isTrashed()) continue;

      // 현재 악보 파일이 들어 있는 폴더가 있으면 그것이 최우선입니다.
      if (currentFileId && folderContainsFileId_(folder, currentFileId)) {
        return folder;
      }

      const updated = folder.getLastUpdated();
      const time = updated ? updated.getTime() : 0;
      if (!newest || time > newestTime) {
        newest = folder;
        newestTime = time;
      }
    } catch (_) {}
  }

  return newest;
}

function getScoreFolder_(createIfMissing) {
  const props = PropertiesService.getScriptProperties();

  // 1) Script Properties에 저장된 ID.
  const savedId = String(props.getProperty(SCORE_STORE.FOLDER_PROP) || '');
  const saved = scoreFolderByIdIfUsable_(savedId);
  if (saved) {
    // 예전 잘못된 폴더 ID가 남아 있어도 canonical 폴더가 살아 있으면 canonical로 복구합니다.
    const canonical = scoreFolderByIdIfUsable_(SCORE_STORE.CANONICAL_FOLDER_ID);
    if (canonical && canonical.getId() !== saved.getId()) {
      props.setProperty(SCORE_STORE.FOLDER_PROP, canonical.getId());
      return canonical;
    }
    return saved;
  }

  // 2) 현재 운영 중인 canonical 폴더. Script Properties가 사라져도 이 폴더를 재사용합니다.
  const canonical = scoreFolderByIdIfUsable_(SCORE_STORE.CANONICAL_FOLDER_ID);
  if (canonical) {
    props.setProperty(SCORE_STORE.FOLDER_PROP, canonical.getId());
    return canonical;
  }

  // 3) canonical 폴더가 삭제된 예외 상황이면 기존 동명 폴더를 먼저 복구합니다.
  const existing = findExistingScoreFolder_();
  if (existing) {
    props.setProperty(SCORE_STORE.FOLDER_PROP, existing.getId());
    return existing;
  }

  if (!createIfMissing) return null;

  // 4) 저장된 ID, canonical, 동명 기존 폴더가 모두 없을 때에만 새 폴더를 만듭니다.
  const folder = DriveApp.createFolder(SCORE_STORE.FOLDER_NAME);
  props.setProperty(SCORE_STORE.FOLDER_PROP, folder.getId());
  return folder;
}


function getAfterScoreFolder_(createIfMissing) {

      }

  }

  }

  if (!createIfMissing) return null;
  const folder = DriveApp.createFolder(AFTER_SCORE_STORE.FOLDER_NAME);
  props.setProperty(AFTER_SCORE_STORE.FOLDER_PROP, folder.getId());
  return folder;
}

function finalizeAfterScore_(newFile, fileName, mimeType) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const props = PropertiesService.getScriptProperties();
    const folder = getAfterScoreFolder_(true);
    const updatedAt = Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone() || 'Asia/Seoul',
      'yyyy-MM-dd HH:mm:ss'
    );
    props.setProperties({
      [AFTER_SCORE_STORE.FILE_PROP]: newFile.getId(),
      [AFTER_SCORE_STORE.NAME_PROP]: fileName || newFile.getName(),
      [AFTER_SCORE_STORE.MIME_PROP]: mimeType || newFile.getMimeType(),
      [AFTER_SCORE_STORE.UPDATED_PROP]: updatedAt
    }, true);

    const files = folder.getFiles();
    while (files.hasNext()) {
      const oldFile = files.next();
      if (oldFile.getId() !== newFile.getId()) {
        try { oldFile.setTrashed(true); } catch (_) {}
      }
    }
    return updatedAt;
  } finally {
    lock.releaseLock();
  }
}

function deleteAfterScore_() {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const folder = getAfterScoreFolder_(false);
    if (folder) {
      const files = folder.getFiles();
      while (files.hasNext()) {
        try { files.next().setTrashed(true); } catch (_) {}
      }
    }
    clearAfterScoreProps_();
  } finally {
    lock.releaseLock();
  }
}

function uploadAfterScoreDirect_(p) {
  cleanupStaleScoreUploads_();

  const fileName = sanitizeScoreFileName_(String(p.fileName || '').trim());
  const mimeType = normalizeScoreMime_(String(p.mimeType || ''), fileName);
  const totalSize = Number(p.totalSize || 0);
  const dataBase64 = String(p.dataBase64 || '');

  validateScoreUpload_(fileName, mimeType, totalSize);
  if (totalSize > SCORE_STORE.DIRECT_MAX_BYTES) {
    throw new Error('직접 업로드 허용 크기를 초과했습니다. 분할 업로드를 사용해주세요.');
  }
  if (!dataBase64) throw new Error('파일 데이터가 비어 있습니다.');

  const bytes = Utilities.base64Decode(dataBase64);
  if (bytes.length !== totalSize) throw new Error('파일 크기 정보가 일치하지 않습니다.');

  const folder = getAfterScoreFolder_(true);
  const blob = Utilities.newBlob(bytes, mimeType, fileName);
  const file = folder.createFile(blob);

  try {
    makeScoreFilePublic_(file);
    const updatedAt = finalizeAfterScore_(file, fileName, mimeType);
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: true,
      transport: 'direct',
      receivedBytes: totalSize,
      file: buildScoreMeta_(file, updatedAt)
    };
  } catch (err) {
    try { file.setTrashed(true); } catch (_) {}
    throw err;
  }
}

function startAfterScoreUpload_(p) {
  cleanupStaleScoreUploads_();

  const fileName = sanitizeScoreFileName_(String(p.fileName || '').trim());
  const mimeType = normalizeScoreMime_(String(p.mimeType || ''), fileName);
  const totalSize = Number(p.totalSize || 0);
  validateScoreUpload_(fileName, mimeType, totalSize);

  const folder = getAfterScoreFolder_(true);
  const initUrl =
    'https://www.googleapis.com/upload/drive/v3/files' +
    '?uploadType=resumable&fields=id,name,mimeType,size';

  const response = UrlFetchApp.fetch(initUrl, {
    method: 'post',
    contentType: 'application/json; charset=UTF-8',
    headers: {
      Authorization: 'Bearer ' + ScriptApp.getOAuthToken(),
      'X-Upload-Content-Type': mimeType,
      'X-Upload-Content-Length': String(totalSize)
    },
    payload: JSON.stringify({
      name: fileName,
      parents: [folder.getId()]
    }),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  if (status < 200 || status >= 300) {
    const detail = truncate_(response.getContentText(), 700);
    throw new Error('예배 후 악보 Drive 업로드 준비 실패 (' + status + '): ' + detail);
  }

  const sessionUrl = String(
    getHeaderIgnoreCase_(response.getAllHeaders(), 'Location') || ''
  );
  if (!sessionUrl) throw new Error('Drive 업로드 세션 URL을 받지 못했습니다.');

  const uploadId = Utilities.getUuid();
  const meta = {
    uploadId: uploadId,
    sessionUrl: sessionUrl,
    fileName: fileName,
    mimeType: mimeType,
    totalSize: totalSize,
    nextStart: 0,
    createdAt: Date.now(),
    complete: false,
    fileId: '',
    updatedAt: ''
  };

  PropertiesService.getScriptProperties().setProperty(
    AFTER_SCORE_STORE.UPLOAD_PREFIX + uploadId,
    JSON.stringify(meta)
  );

  return {
    ok: true,
    service: SCORE_STORE.APP,
    version: SCORE_API_VERSION,
    transport: 'resumable',
    uploadId: uploadId,
    chunkSize: SCORE_STORE.CHUNK_SIZE,
    totalSize: totalSize
  };
}

function uploadAfterScoreChunk_(p) {
  const uploadId = String(p.uploadId || '');
  if (!uploadId) throw new Error('업로드 세션 정보가 없습니다.');

  const props = PropertiesService.getScriptProperties();
  const key = AFTER_SCORE_STORE.UPLOAD_PREFIX + uploadId;
  const raw = props.getProperty(key);
  if (!raw) throw new Error('업로드 세션이 만료되었습니다. 처음부터 다시 업로드해주세요.');

  let meta;
  try {
    meta = JSON.parse(raw);
  } catch (_) {
    props.deleteProperty(key);
    throw new Error('업로드 세션 정보가 손상되었습니다. 다시 업로드해주세요.');
  }

  if (meta.complete && meta.fileId) {
    const completedFile = DriveApp.getFileById(String(meta.fileId));
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: true,
      transport: 'resumable',
      receivedBytes: Number(meta.totalSize || 0),
      file: buildScoreMeta_(completedFile, String(meta.updatedAt || ''))
    };
  }

  const start = Number(p.start || 0);
  const end = Number(p.end || -1);
  const totalSize = Number(p.totalSize || meta.totalSize || 0);
  const dataBase64 = String(p.dataBase64 || '');

  if (!dataBase64) throw new Error('업로드 조각 데이터가 비어 있습니다.');
  if (totalSize !== Number(meta.totalSize)) throw new Error('파일 크기 정보가 일치하지 않습니다.');
  if (start < 0 || end < start || end >= totalSize) throw new Error('업로드 범위가 올바르지 않습니다.');

  const expectedNext = Number(meta.nextStart || 0);
  if (end < expectedNext) {
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: false,
      transport: 'resumable',
      receivedBytes: expectedNext
    };
  }
  if (start !== expectedNext) {
    throw new Error('업로드 순서가 어긋났습니다. 예상 시작 위치: ' + expectedNext + ', 받은 위치: ' + start);
  }

  const bytes = Utilities.base64Decode(dataBase64);
  if (bytes.length !== end - start + 1) throw new Error('업로드 조각 크기가 일치하지 않습니다.');

  const response = UrlFetchApp.fetch(String(meta.sessionUrl), {
    method: 'put',
    contentType: String(meta.mimeType || 'application/octet-stream'),
    headers: {
      'Content-Range': 'bytes ' + start + '-' + end + '/' + totalSize
    },
    payload: bytes,
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  if (status === 308) {
    meta.nextStart = end + 1;
    props.setProperty(key, JSON.stringify(meta));
    return {
      ok: true,
      service: SCORE_STORE.APP,
      version: SCORE_API_VERSION,
      complete: false,
      transport: 'resumable',
      receivedBytes: meta.nextStart
    };
  }

  if (status >= 200 && status < 300) {
    let driveMeta = {};
    try { driveMeta = JSON.parse(response.getContentText() || '{}'); } catch (_) {}
    const fileId = String(driveMeta.id || '');
    if (!fileId) throw new Error('Drive 업로드는 완료됐지만 파일 ID를 받지 못했습니다.');

    const file = DriveApp.getFileById(fileId);
    try {
      makeScoreFilePublic_(file);
      const updatedAt = finalizeAfterScore_(
        file,
        String(meta.fileName || file.getName()),
        String(meta.mimeType || file.getMimeType())
      );
      meta.complete = true;
      meta.fileId = fileId;
      meta.updatedAt = updatedAt;
      meta.nextStart = totalSize;
      props.setProperty(key, JSON.stringify(meta));

      return {
        ok: true,
        service: SCORE_STORE.APP,
        version: SCORE_API_VERSION,
        complete: true,
        transport: 'resumable',
        receivedBytes: totalSize,
        file: buildScoreMeta_(file, updatedAt)
      };
    } catch (err) {
      try { file.setTrashed(true); } catch (_) {}
      throw err;
    }
  }

  throw new Error(
    '예배 후 악보 Drive 업로드 실패 (' + status + '): ' +
    truncate_(response.getContentText(), 500)
  );
}

function normalizeAfterYoutubeLinks_(rawLinks) {
  const input = Array.isArray(rawLinks) ? rawLinks : [];
  const out = [];
  const seen = {};
  input.slice(0, 30).forEach(function(value) {
    const url = String(value || '').trim();
    if (!url || url.length > 500) return;
    if (!/^https?:\/\/(?:www\.)?(?:youtube\.com|youtu\.be)\//i.test(url) &&
        !/^https?:\/\/(?:www\.)?youtube-nocookie\.com\//i.test(url)) return;
    if (seen[url]) return;
    seen[url] = true;
    out.push(url);
  });
  return out;
}

function readAfterYoutubeLinksFile_() {
  const folder = getAfterLinkFolder_(false);
  if (!folder) return null;

  const files = folder.getFilesByName(AFTER_LINK_STORE.FILE_NAME);
  let newest = null;
  let newestTime = -1;
  while (files.hasNext()) {
    const file = files.next();
    try {
      if (file.isTrashed()) continue;
      const time = file.getLastUpdated().getTime();
      if (time > newestTime) {
        newest = file;
        newestTime = time;
      }
    } catch (_) {}
  }
  if (!newest) return null;

  try {
    const parsed = JSON.parse(newest.getBlob().getDataAsString('UTF-8') || '{}');
    const values = Array.isArray(parsed) ? parsed : parsed.links;
    if (!Array.isArray(values)) return null;
    return normalizeAfterYoutubeLinks_(values);
  } catch (_) {
    return null;
  }
}

function writeAfterYoutubeLinksFile_(links) {
  const folder = getAfterLinkFolder_(true);
  const payload = JSON.stringify({
    version: 1,
    updatedAt: new Date().toISOString(),
    links: normalizeAfterYoutubeLinks_(links)
  });

  // 새 파일을 먼저 만든 뒤 구 파일을 정리해 저장 중 실패 시 기존 데이터가 남도록 합니다.
  const newFile = folder.createFile(
    AFTER_LINK_STORE.FILE_NAME,
    payload,
    MimeType.PLAIN_TEXT
  );

  const files = folder.getFilesByName(AFTER_LINK_STORE.FILE_NAME);
  while (files.hasNext()) {
    const file = files.next();
    if (file.getId() === newFile.getId()) continue;
    try { file.setTrashed(true); } catch (_) {}
  }
  return newFile.getId();
}

function getAfterYoutubeLinks_() {
  // Drive 파일을 영구 원본으로 우선 사용합니다.
  const fromDrive = readAfterYoutubeLinksFile_();
  if (fromDrive !== null) {
    PropertiesService.getScriptProperties().setProperty(
      AFTER_SCORE_STORE.YOUTUBE_LINKS_PROP,
      JSON.stringify(fromDrive)
    );
    return fromDrive;
  }

  // 기존 v3.12.8 Script Properties 데이터가 있으면 읽고 Drive로 자동 마이그레이션합니다.
  const raw = PropertiesService.getScriptProperties().getProperty(AFTER_SCORE_STORE.YOUTUBE_LINKS_PROP);
  if (!raw) return [];

  try {
    const links = normalizeAfterYoutubeLinks_(JSON.parse(raw));
    try { writeAfterYoutubeLinksFile_(links); } catch (_) {}
    return links;
  } catch (_) {
    return [];
  }
}

function setAfterYoutubeLinks_(linksJson) {
  let parsed = [];
  try {
    parsed = JSON.parse(String(linksJson || '[]'));
  } catch (_) {
    throw new Error('YouTube 링크 목록 형식이 올바르지 않습니다.');
  }

  const links = normalizeAfterYoutubeLinks_(parsed);
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    // Properties + Drive JSON 두 곳 모두 기록합니다.
    PropertiesService.getScriptProperties().setProperty(
      AFTER_SCORE_STORE.YOUTUBE_LINKS_PROP,
      JSON.stringify(links)
    );
    writeAfterYoutubeLinksFile_(links);

    // 기록 직후 영구 원본을 다시 읽어 실제 저장값까지 검증합니다.
    const verified = readAfterYoutubeLinksFile_();
    if (verified === null || JSON.stringify(verified) !== JSON.stringify(links)) {
      throw new Error('예배 후 YouTube 링크 저장 검증에 실패했습니다.');
    }
    return verified;
  } finally {
    lock.releaseLock();
  }
}

function cleanupStaleScoreUploads_() {
  const props = PropertiesService.getScriptProperties();
  const all = props.getProperties();
  const cutoff = Date.now() - SCORE_STORE.STALE_UPLOAD_MS;

  Object.keys(all).forEach(function(key) {
    if (key.indexOf(SCORE_STORE.UPLOAD_PREFIX) !== 0 &&
        key.indexOf(AFTER_SCORE_STORE.UPLOAD_PREFIX) !== 0) return;
    try {
      const meta = JSON.parse(all[key] || '{}');
      if (!meta.createdAt || Number(meta.createdAt) < cutoff) {
        props.deleteProperty(key);
      }
    } catch (_) {
      props.deleteProperty(key);
    }
  });
}

function scoreReply_(data, requestId) {
  const payload = Object.assign({
    source: 'worship-score-api',
    requestId: String(requestId || ''),
    service: SCORE_STORE.APP,
    version: SCORE_API_VERSION
  }, data || {});

  const json = JSON.stringify(payload)
    .replace(/</g, '\u003c')
    .replace(/>/g, '\u003e')
    .replace(/&/g, '\u0026');

  const html =
    '<!doctype html><html><head><meta charset="utf-8"></head><body>' +
    '<script>' +
    'var m=' + json + ';' +
    'try{window.top.postMessage(m,"*");}catch(e){}' +
    'try{window.parent.postMessage(m,"*");}catch(e){}' +
    '<\/script>' +
    '</body></html>';

  return HtmlService
    .createHtmlOutput(html)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function scoreTextOutput_(data, callback) {
  const json = JSON.stringify(data || {});
  if (callback) {
    const safe = String(callback).replace(/[^a-zA-Z0-9_$\.]/g, '');
    if (safe) {
      return ContentService
        .createTextOutput(safe + '(' + json + ');')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
  }
  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

function sanitizeScoreFileName_(name) {
  return String(name || '').replace(/[\\/:*?"<>|]/g, '_').slice(0, 180);
}

function normalizeScoreMime_(mimeType, fileName) {
  const ext = String(fileName || '').toLowerCase().split('.').pop();
  const map = {
    pdf: 'application/pdf',
    ppt: 'application/vnd.ms-powerpoint',
    pptx: 'application/vnd.openxmlformats-officedocument.presentationml.presentation',
    doc: 'application/msword',
    docx: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    jpg: 'image/jpeg',
    jpeg: 'image/jpeg',
    png: 'image/png'
  };
  return map[ext] || mimeType || 'application/octet-stream';
}

function isAllowedScoreFile_(fileName, mimeType) {
  const ext = String(fileName || '').toLowerCase().split('.').pop();
  return ['pdf','ppt','pptx','doc','docx','jpg','jpeg','png'].indexOf(ext) !== -1 && !!mimeType;
}

function getHeaderIgnoreCase_(headers, name) {
  const target = String(name || '').toLowerCase();
  const keys = Object.keys(headers || {});
  for (let i = 0; i < keys.length; i++) {
    if (String(keys[i]).toLowerCase() === target) {
      const value = headers[keys[i]];
      return Array.isArray(value) ? value[0] : value;
    }
  }
  return '';
}

function truncate_(text, maxLen) {
  const s = String(text || '');
  return s.length > maxLen ? s.slice(0, maxLen) + '…' : s;
}

function errorMessage_(err) {
  return String(err && err.message ? err.message : err);
}
